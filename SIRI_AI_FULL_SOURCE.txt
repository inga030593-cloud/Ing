import { handle } from "hono/aws-lambda";
import { timingSafeEqual } from "node:crypto";
import app from "./src/index.js";
import { _bindVisitor, _mintedRealtimeToken } from "./src/pipe.js";
app.onError((err, c) => {
  if (String(err?.message) === "rate_limited") return c.json({ error: "rate_limited" }, 429);
  console.error(err);
  return c.text("Internal Server Error", 500);
});
const _inner = handle(app);
function _eq(a, b) {
  const x = Buffer.from(String(a ?? "")), y = Buffer.from(String(b ?? ""));
  return x.length === y.length && timingSafeEqual(x, y);
}
// cf-connecting-ip first; else last X-Forwarded-For hop (closest trusted).
function _realIp(h) {
  if (h["cf-connecting-ip"]) return String(h["cf-connecting-ip"]).trim();
  const fwd = h["x-forwarded-for"];
  if (!fwd) return "";
  const parts = String(fwd).split(",");
  return parts[parts.length - 1].trim();
}
function _cache(event, res, authed) {
  if (!res) return res;
  const headers = res.headers || (res.headers = {});
  // A realtime token lives an hour: whatever carries one is never edge-cached, whatever the app declared.
  if (_mintedRealtimeToken()) {
    for (const k of Object.keys(headers)) if (k.toLowerCase() === "cache-control") delete headers[k];
    headers["cache-control"] = "private, no-store";
    return res;
  }
  if (event?.requestContext?.http?.method !== "GET") return res;
  if (Object.keys(headers).some((k) => k.toLowerCase() === "cache-control")) return res;
  // Cookie / Set-Cookie / restored Authorization → private; else edge-cacheable.
  const priv = !!event.headers?.cookie || (Array.isArray(res.cookies) && res.cookies.length > 0) || authed;
  headers["cache-control"] = res.statusCode === 200 && !priv ? "public, max-age=0, must-revalidate, s-maxage=86400" : "private, no-store";
  return res;
}
export const handler = async (event, context) => {
  const h = event?.headers || {};
  const edgeOk = _eq(h["x-edge-key"], process.env.EDGE_KEY);
  const serverOk = _eq(h["x-origin-secret"], process.env.PIPE_ORIGIN_SECRET) && _eq(h["x-pipe-key"], process.env.PIPE_KEY);
  if (!edgeOk && !serverOk) return { statusCode: 403, headers: { "content-type": "text/plain" }, body: "forbidden" };
  // SigV4 proxy overwrites Authorization; nginx stashed the visitor value as X-Original-Authorization.
  const authed = !!h["x-original-authorization"];
  if (authed) h["authorization"] = h["x-original-authorization"];
  else delete h["authorization"];
  for (const k of ["x-original-authorization", "x-edge-key", "x-amz-security-token", "x-amz-date", "x-amz-content-sha256"]) delete h[k];
  // POST keeps origin/pipe headers for pipe.isScheduler(); GET strips them for cache safety.
  if (event?.requestContext?.http?.method !== "POST") { delete h["x-origin-secret"]; delete h["x-pipe-key"]; }
  _bindVisitor(_realIp(h));
  return _cache(event, await _inner(event, context), authed);
};

import pipe from "../pipe.js";
import { noStore } from "../lib/visitor.js";
import { streamCloud, trimHistory, buildSystem, buildMessages, CONTEXT_CHOICES } from "../lib/cloud.js";
import { identity, isMine, ownChat, ownerFields, ownerWhere, ID_RE } from "../lib/access.js";
import { getSettings } from "../lib/settings.js";

const HOUR = 60 * 60 * 1000;
const MAX_PER_HOUR = 120;
// Групповой диалог Смит ↔ Сири: максимум 2 раунда обмена репликами сверх ответов.
const MAX_GROUP_ROUNDS = 2;

function cleanTitle(text) {
  const t = text.replace(/\s+/g, " ").trim();
  return t.length > 48 ? `${t.slice(0, 48).trim()}…` : t || "Новый чат";
}

function unavailable(c) {
  return c.json({ error: "unavailable" }, 503);
}

function contextTokens(choice) {
  const found = CONTEXT_CHOICES.find((item) => item.key === choice);
  return found ? found.tokens : 1_000_000;
}

// Память между чатами: строки из других разговоров плюс оценки ответов.
function buildMemory(rows, chatId) {
  const past = rows.filter((m) => m.data.chat_id !== chatId && m.data.kind !== "image" && m.data.text);
  const memory = past
    .slice(0, 8)
    .reverse()
    .map((m) => {
      const who = m.data.role === "ai" ? (PERSONAS[m.data.who]?.name || "Siri") : "Пользователь";
      return `- ${who}: ${String(m.data.text).replace(/\s+/g, " ").slice(0, 160)}`;
    });
  const rated = rows.filter((m) => m.data.role === "ai" && m.data.rating);
  const liked = rated.filter((m) => m.data.rating === "up").slice(0, 2);
  const disliked = rated.filter((m) => m.data.rating === "down").slice(0, 2);
  return {
    memory: memory.join("\n"),
    liked: liked.map((m) => String(m.data.text).replace(/\s+/g, " ").slice(0, 220)).join("\n---\n"),
    disliked: disliked.map((m) => String(m.data.text).replace(/\s+/g, " ").slice(0, 220)).join("\n---\n"),
  };
}

async function shieldCheck(c, who, text) {
  const settings = await getSettings(who);
  if (!settings.shield) return null;
  const state = await securityState(who);
  if (state.blockedUntil > Date.now()) {
    return c.json({ error: "shield_blocked", retry_after: Math.ceil((state.blockedUntil - Date.now()) / 1000) }, 403);
  }
  const reason = inspectText(text);
  if (reason) {
    const next = await noteAttack(who, reason);
    return c.json(
      { error: "shield_blocked", retry_after: next.blockedUntil ? Math.ceil((next.blockedUntil - Date.now()) / 1000) : 0 },
      403,
    );
  }
  if (await shieldRateExceeded(who)) return c.json({ error: "shield_rate_limited" }, 429);
  return null;
}

export function RegisterApi(app) {
  app.get("/api/chats", async (c) => {
    noStore(c);
    const who = await identity(c);
    const where = ownerWhere(who);
    let chats = await pipe.data.list("chats", {
      where,
      sort: "updated_at",
      order: "desc",
      limit: 50,
    });
    if (!who.user) chats = chats.filter((ch) => !ch.data.user_id);
    return c.json({
      chats: chats.map((ch) => ({
        id: ch._id,
        title: ch.data.title,
        preview: ch.data.preview || "",
        mode: ch.data.mode || "siri",
        updated_at: ch.data.updated_at,
      })),
    });
  });

  app.get("/api/chats/:id", async (c) => {
    noStore(c);
    const who = await identity(c);
    const chat = await ownChat(c, who);
    if (!chat) return c.json({ error: "not_found" }, 404);
    const page = await pipe.data.list("messages", {
      where: { chat_id: chat._id },
      sort: "created_at",
      order: "desc",
      limit: 200,
    });
    const messages = page
      .slice()
      .reverse()
      .map((m) => ({
        id: m._id,
        role: m.data.role,
        who: m.data.who || "siri",
        text: m.data.text,
        rating: m.data.rating || "",
        sources: m.data.sources || [],
        reply_to: m.data.reply_to || "",
        created_at: m.data.created_at,
      }));
    return c.json({ chat: { id: chat._id, title: chat.data.title, mode: chat.data.mode || "siri" }, messages });
  });

  app.delete("/api/chats/:id", async (c) => {
    noStore(c);
    const who = await identity(c);
    const chat = await ownChat(c, who);
    if (!chat) return c.json({ error: "not_found" }, 404);
    try {
      for (;;) {
        const batch = await pipe.data.list("messages", {
          where: { chat_id: chat._id },
          limit: 100,
        });
        if (!batch.length) break;
        for (const m of batch) await pipe.data.remove("messages", m._id);
        if (batch.length < 100) break;
      }
      await pipe.data.remove("chats", chat._id);
    } catch (e) {
      if (["quota_exceeded", "rate_limited"].includes(e.message)) return unavailable(c);
      throw e;
    }
    return c.json({ ok: true });
  });

  app.post("/api/chat", async (c) => {
    noStore(c);
    const who = await identity(c);
    const body = await c.req.json().catch(() => null);
    const text = typeof body?.text === "string" ? body.text.trim() : "";
    if (!text) return c.json({ error: "empty" }, 400);
    if (text.length > 4000) return c.json({ error: "too_long" }, 400);
    const model = typeof body?.model === "string" ? body.model : "cyclone";
    const mode = ["siri", "smith", "group"].includes(body?.mode) ? body.mode : "siri";
    // Ответ на конкретное сообщение (кнопка «Ответить»): цитата уходит в промпт.
    const replyQuote =
      typeof body?.reply_to === "string" ? body.reply_to.replace(/\s+/g, " ").slice(0, 500) : "";

    const blocked = await shieldCheck(c, who, text);
    if (blocked) return blocked;

    const now = Date.now();
    const wantedId = typeof body.chat_id === "string" ? body.chat_id : "";
    if (wantedId && !ID_RE.test(wantedId)) return c.json({ error: "not_found" }, 404);

    const [recent, found, settings] = await Promise.all([
      pipe.data
        .count("messages", { where: { ...ownerWhere(who), created_at: { $gt: now - HOUR } } })
        .catch(() => 0),
      wantedId ? pipe.data.get("chats", wantedId) : Promise.resolve(null),
      getSettings(who),
    ]);
    if (recent >= MAX_PER_HOUR) return c.json({ error: "rate_limited" }, 429);
    if (wantedId && !isMine(found, who)) return c.json({ error: "not_found" }, 404);
    let chat = wantedId ? found : null;

    const owner = ownerFields(who);
    try {
      if (!chat) {
        chat = await pipe.data.insert("chats", {
          ...owner,
          title: cleanTitle(text),
          preview: "",
          mode,
          created_at: now,
          updated_at: now,
        });
      }
      await pipe.data.insert("messages", {
        chat_id: chat._id,
        ...owner,
        role: "user",
        kind: "text",
        text,
        reply_to: replyQuote,
        created_at: now,
      });
    } catch (e) {
      if (["quota_exceeded", "rate_limited"].includes(e.message)) return unavailable(c);
      throw e;
    }

    const chatId = chat._id;
    const chatMode = chat.data.mode || mode;

    // История чата и память между чатами не зависят друг от друга — читаем вместе.
    const [page, memoryRows] = await Promise.all([
      pipe.data
        .list("messages", {
          where: { chat_id: chatId },
          sort: "created_at",
          order: "desc",
          limit: 200,
        })
        .catch(() => []),
      pipe.data
        .list("messages", { where: ownerWhere(who), sort: "created_at", order: "desc", limit: 40 })
        .catch(() => []),
    ]);

    const budget = contextTokens(settings.context);
    const context = buildMemory(memoryRows, chatId);
    // Бюджет контекста из настроек применяется по-настоящему: лишняя история отрезается.
    const history = trimHistory(
      page
        .slice()
        .reverse()
        .map((m) => ({
          role: m.data.role === "ai" ? "assistant" : "user",
          who: m.data.who || "siri",
          content: String(m.data.text || ""),
        })),
      budget,
    );

    const speakers = resolveSpeakers(text, chatMode);
    const group = chatMode === "group" || speakers.length > 1;
    const doSearch = wantsSearch(text);

    const encoder = new TextEncoder();
    const replyStream = new ReadableStream({
      async start(controller) {
        const send = (event) => {
          try {
            controller.enqueue(encoder.encode(`data: ${JSON.stringify(event)}\n\n`));
          } catch {
            /* посетитель закрыл страницу */
          }
        };
        send({ type: "start", chat: { id: chatId, title: chat.data.title, mode: chatMode }, speakers, search: doSearch });

        // Интернет-поиск: собираем источники до обращения к модели.
        let search = null;
        if (doSearch) {
          send({ type: "status", text: "Ищу в интернете…" });
          search = await searchWeb(searchQuery(text) || text);
          send({ type: "sources", sources: search.results, query: search.query });
        }

        // Один общий список «виртуальной истории»: сюда добавляются свежие реплики,
        // поэтому Смит видит ответ Сиры, а Сири — ответ Смита.
        const script = history.slice();

        const answer = async (personaKey) => {
          const persona = PERSONAS[personaKey];
          send({ type: "who", who: personaKey, name: persona.name });
          const system = buildSystem({
            personaKey,
            group,
            bio: settings.bio,
            memory: context.memory,
            liked: context.liked,
            disliked: context.disliked,
            search,
          });
          let messages = buildMessages({ history: script, personaKey, group });
          if (replyQuote) {
            messages = [
              ...messages,
              { role: "user", content: `(Пользователь отвечает на сообщение: «${replyQuote}»)\n${text}` },
            ];
          }
          const outgoing = [{ role: "system", content: system }, ...messages];

          let stream;
          try {
            stream = await streamCloud(outgoing, model);
          } catch {
            stream = { error: "upstream" };
          }
          if (stream.error) {
            send({ type: "error", who: personaKey, error: "cloud_unavailable" });
            return "";
          }

          let full = "";
          try {
            for await (const delta of stream.deltas) {
              full += delta;
              send({ type: "delta", who: personaKey, text: delta });
            }
          } catch {
            /* поток оборвался — сохраняем то, что пришло */
          }
          full = full.trim();
          if (!full) {
            send({ type: "error", who: personaKey, error: "cloud_unavailable" });
            return "";
          }

          script.push({ role: "assistant", who: personaKey, content: full });
          try {
            const reply = await pipe.data.insert("messages", {
              chat_id: chatId,
              ...owner,
              role: "ai",
              kind: "text",
              who: personaKey,
              text: full,
              sources: search?.results || [],
              created_at: Date.now(),
            });
            send({ type: "saved", who: personaKey, reply: { id: reply._id, who: personaKey, text: full, sources: search?.results || [] } });
          } catch {
            send({ type: "error", who: personaKey, error: "save_failed" });
          }
          return full;
        };

        // Основные ответы: Siri и/или Смит по очереди.
        for (const personaKey of speakers) {
          await answer(personaKey);
        }

        // Просьба «поговорите о…» в AI Group: Смит и Сири обмениваются репликами.
        if (group && wantsDiscussion(text)) {
          let last = speakers[speakers.length - 1] || "siri";
          for (let round = 0; round < MAX_GROUP_ROUNDS; round += 1) {
            const next = last === "siri" ? "smith" : "siri";
            const text2 = await answer(next);
            if (!text2) break;
            last = next;
          }
        }

        try {
          await pipe.data.update("chats", chatId, {
            data: {
              preview: (script[script.length - 1]?.content || text).replace(/\s+/g, " ").slice(0, 120),
              updated_at: Date.now(),
            },
          });
        } catch {
          /* превью не критично */
        }
        send({ type: "done" });
        controller.close();
      },
    });

    return new Response(replyStream, {
      headers: {
        "content-type": "text/event-stream; charset=utf-8",
        "cache-control": "private, no-store, no-transform",
        "x-accel-buffering": "no",
      },
    });
  });
}

function settingsView(settings, security) {

function safeFilename(title) {
  const base = String(title || "chat").replace(/[^\p{L}\p{N}\-_ ]+/gu, "").replace(/\s+/g, "-").slice(0, 60);
  return (base || "chat") + ".md";
}

function exportMarkdown(chat, messages) {
  const mode = chat.data.mode === "group" ? "AI Group" : chat.data.mode === "smith" ? "Smith" : "Siri";
  const lines = [`# ${chat.data.title}`, "", `Экспорт из Siri AI · Режим: ${mode} · ${new Date().toLocaleString("ru-RU")}`, ""];
  for (const m of messages) {
    const who =
      m.data.role === "ai" ? (PERSONAS[m.data.who]?.name || "Siri") : "Вы";
    lines.push(`**${who}:** ${m.data.text}`, "");
    for (const source of m.data.sources || []) lines.push(`- Источник: ${source.url}`, "");
  }
  return lines.join("\n");
}

        .map((m) => ({ role: m.data.role === "ai" ? "assistant" : "user", content: m.data.text }));
      history.push({ role: "user", content: text.slice(0, 4000) });

      await pipe.data.insert("messages", { chat_id: tgChatId, telegram: true, role: "user", text: text.slice(0, 4000), created_at: since }).catch(() => {});
      const answer = await askCloud(history);
      const reply = answer.text || "Сейчас облако не отвечает. Попробуй ещё раз через минуту.";
      if (answer.text) {
        await pipe.data.insert("messages", { chat_id: tgChatId, telegram: true, role: "ai", who: "siri", text: answer.text, created_at: Date.now() }).catch(() => {});
      }
      await sendTelegram(token, chatId, reply);
      return c.json({ ok: true });
    } catch {
      return c.json({ ok: true });
    }
  });
}

export function RegisterAccountPages(app) {
  app.get("/login", async (c) => {
    noStore(c);
    if (await currentUser(c)) return c.redirect("/");
    const code = c.req.query("e");
    return c.html(<LoginPage origin={pipe.origin(c.req)} error={LOGIN_ERRORS[code] || ""} />);
  });

  app.post("/login", async (c) => {
    noStore(c);
    const body = await c.req.parseBody();
    const email = normalizeEmail(body.email);
    const password = String(body.password || "");
    if (!validEmail(email) || !password) return c.redirect("/login?e=invalid");
    try {
      if (await tooManyAttempts(email)) return c.redirect("/login?e=locked");
      const user = await checkLogin(email, password);
      if (!user) {
        await recordFailedAttempt(email).catch(() => {});
        return c.redirect("/login?e=bad");
      }
      await startSession(c, user);
      await claimVisitorChats(visitorId(c), user._id);
    } catch (e) {
      if (["quota_exceeded", "rate_limited"].includes(e.message)) return c.redirect("/login?e=unavailable");
      throw e;
    }
    return c.redirect("/");
  });

  app.get("/signup", async (c) => {
    noStore(c);
    if (await currentUser(c)) return c.redirect("/");
    const code = c.req.query("e");
    return c.html(<SignupPage origin={pipe.origin(c.req)} error={SIGNUP_ERRORS[code] || ""} />);
  });

  app.post("/signup", async (c) => {
    noStore(c);
    const body = await c.req.parseBody();
    const email = normalizeEmail(body.email);
    const password = String(body.password || "");
    const name = String(body.name || "").replace(/\s+/g, " ").trim().slice(0, 60);
    if (!validEmail(email)) return c.redirect("/signup?e=email");
    if (password.length < 8 || password.length > 200) return c.redirect("/signup?e=short");

    try {
      const [existing] = await pipe.data.list("users", { where: { email }, limit: 1 });
      if (existing) return c.redirect("/signup?e=exists");
      const user = await pipe.data.insert("users", {
        email,
        name,
        password_hash: hashPassword(password),
        role: "user",
        created_at: Date.now(),
      });
      await startSession(c, user);
      await claimVisitorChats(visitorId(c), user._id);
    } catch (e) {
      if (["quota_exceeded", "rate_limited"].includes(e.message)) return c.redirect("/signup?e=unavailable");
      throw e;
    }
    return c.redirect("/");
  });

  app.get("/account", async (c) => {
    noStore(c);
    const user = await currentUser(c);
    if (!user) return c.redirect("/login");
    return c.html(<AccountPage origin={pipe.origin(c.req)} user={user} />);
  });

  app.post("/logout", async (c) => {
    noStore(c);
    try {
      await endSession(c);
    } catch (e) {
      if (!["quota_exceeded", "rate_limited"].includes(e.message)) throw e;
    }
    return c.redirect("/");
  });
}

export const ID_RE = /^[a-f0-9]{24}$/;

// Who is talking: a signed-in account, or this browser when signed out.
export async function identity(c) {
  const user = await currentUser(c);
  return { user, vid: visitorId(c) };
}

export function isMine(record, who) {
  if (!record) return false;
  if (who.user) return record.data.user_id === who.user._id;
  return !record.data.user_id && record.data.visitor_id === who.vid;
}

export async function ownChat(c, who) {
  const id = c.req.param("id");
  if (!ID_RE.test(id)) return null;
  const chat = await pipe.data.get("chats", id);
  return isMine(chat, who) ? chat : null;
}

export function ownerFields(who) {
  return who.user ? { visitor_id: who.vid, user_id: who.user._id } : { visitor_id: who.vid };
}

// Matches records (chats and messages) that belong to the current visitor.
export function ownerWhere(who) {
  return who.user ? { user_id: who.user._id } : { visitor_id: who.vid };
}

import { scryptSync, randomBytes, timingSafeEqual } from "node:crypto";
import { getCookie, setCookie, deleteCookie } from "hono/cookie";
import pipe from "../pipe.js";

const SESSION_DAYS = 30;
const COOKIE = "sid";

export function normalizeEmail(value) {
  return String(value || "").trim().toLowerCase();
}

export function validEmail(email) {
  return email.length <= 254 && /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(email);
}

export function hashPassword(password) {
  const salt = randomBytes(16);
  return `${salt.toString("hex")}:${scryptSync(password, salt, 32).toString("hex")}`;
}

export function verifyPassword(password, stored) {
  const [salt, hash] = String(stored || "").split(":");
  if (!salt || !hash) return false;
  const test = scryptSync(String(password), Buffer.from(salt, "hex"), 32);
  const want = Buffer.from(hash, "hex");
  return want.length === test.length && timingSafeEqual(want, test);
}

export function displayName(user) {
  return user?.data?.name || String(user?.data?.email || "").split("@")[0] || "Аккаунт";
}

// The user for an email and password, or null.
export async function checkLogin(email, password) {
  const e = normalizeEmail(email);
  const [u] = await pipe.data.list("users", { where: { email: e }, limit: 1 });
  return u && verifyPassword(password, u.data.password_hash) ? u : null;
}

export async function startSession(c, user) {
  const token = randomBytes(24).toString("hex");
  await pipe.data.insert("sessions", {
    token,
    user_id: user._id,
    expires_at: Date.now() + SESSION_DAYS * 86_400_000,
    created_at: Date.now(),
  });
  setCookie(c, COOKIE, token, {
    httpOnly: true,
    secure: true,
    sameSite: "Lax",
    path: "/",
    maxAge: SESSION_DAYS * 86_400,
  });
}

// The signed-in user or null. Call it at the top of every protected route.
export async function currentUser(c) {
  const token = getCookie(c, COOKIE);
  if (!token) return null;
  const [session] = await pipe.data.list("sessions", { where: { token }, limit: 1 });
  if (!session) return null;
  if (session.data.expires_at < Date.now()) {
    try {
      await pipe.data.remove("sessions", session._id);
    } catch {
      // The expired session stays until the next sweep; it is already unusable.
    }
    return null;
  }
  return pipe.data.get("users", session.data.user_id);
}

export async function endSession(c) {
  const token = getCookie(c, COOKIE);
  if (token) {
    const [session] = await pipe.data.list("sessions", { where: { token }, limit: 1 });
    if (session) await pipe.data.remove("sessions", session._id);
  }
  deleteCookie(c, COOKIE, { path: "/" });
}

// Link the chats this browser made before signing in to the account, so nothing is lost.
export async function claimVisitorChats(visitorId, userId) {
  if (!visitorId || !userId) return;
  try {
    const chats = await pipe.data.list("chats", { where: { visitor_id: visitorId }, limit: 50 });
    for (const chat of chats) {
      if (!chat.data.user_id) await pipe.data.update("chats", chat._id, { data: { user_id: userId } });
    }
  } catch {
    // Best effort: the account still works if an old chat cannot be linked.
  }
}

// Flood limit: 5 failed logins per email in 15 minutes.
const WINDOW_MS = 15 * 60_000;

export async function tooManyAttempts(email) {
  const [a] = await pipe.data.list("login_attempts", { where: { email }, limit: 1 });
  return !!a && a.data.count >= 5 && Date.now() - a.data.window_start < WINDOW_MS;
}

export async function recordFailedAttempt(email) {
  const now = Date.now();
  const [a] = await pipe.data.list("login_attempts", { where: { email }, limit: 1 });
  if (!a) return pipe.data.insert("login_attempts", { email, count: 1, window_start: now });
  if (now - a.data.window_start >= WINDOW_MS) {
    return pipe.data.update("login_attempts", a._id, { data: { count: 1, window_start: now } });
  }
  return pipe.data.update("login_attempts", a._id, { inc: { count: 1 } });
}

  { key: "50m", tokens: 50_000_000, label: "50 млн токенов" },
];

  return MODELS[key] || MODELS.cyclone;
}

// Грубая оценка токенов: ~4 символа на токен для русского и английского.
function tokenEstimate(text) {
  return Math.ceil(String(text || "").length / 4);
}

// Бюджет контекста из настроек реально применяется: в историю попадают только
// те сообщения, что укладываются в выбранный лимит токенов.
export function trimHistory(history, budgetTokens) {
  const limit = Math.max(4_000, Number(budgetTokens) || 1_000_000);
  const kept = [];
  let used = 0;
  for (let i = history.length - 1; i >= 0; i -= 1) {
    const cost = tokenEstimate(history[i].content) + 8;
    if (used + cost > limit) break;
    used += cost;
    kept.unshift(history[i]);
  }
  return kept;
}

// Системная инструкция: персона + групповой режим + биография + память + поиск.
export function buildSystem({ personaKey, group, bio, memory, liked, disliked, search }) {
  const persona = PERSONAS[personaKey] || PERSONAS.siri;
  const parts = [persona.prompt];
  if (group) parts.push(groupPrompt(personaKey));
  if (bio) {
    parts.push(
      "Персонализированная память о пользователе (его биография из настроек). Учитывай её в ответах, не пересказывай без необходимости:\n" +
        String(bio).slice(0, 4000),
    );
  }
  if (memory) {
    parts.push(
      "Память о прошлых диалогах этого пользователя (другие чаты). Используй её, когда она уместна, не пересказывай без просьбы:\n" +
        memory,
    );
  }
  if (liked) parts.push("Пользователь высоко оценил такие ответы — отвечай в похожем духе:\n" + liked);
  if (disliked) parts.push("Эти ответы пользователю не понравились — не повторяй их стиль и ошибки:\n" + disliked);
  const web = searchContext(search);
  if (web) parts.push(web);
  return parts.join("\n\n");
}

// История группового чата с точки зрения одной персоны:
// свои ответы — assistant, чужие ответы ИИ — user с префиксом имени.
export function buildMessages({ history, personaKey, group }) {
  const persona = PERSONAS[personaKey] || PERSONAS.siri;
  const out = [];
  for (const item of history) {
    if (item.role === "assistant") {
      if (item.who === personaKey || !group) {
        out.push({ role: "assistant", content: item.content });
      } else {
        const other = PERSONAS[item.who] || { name: "AI" };
        out.push({ role: "user", content: `[${other.name}]: ${item.content}` });
      }
    } else {
      out.push({ role: "user", content: item.content });
    }
  }
  return out;
}

function requestHeaders() {
  return {
    "content-type": "application/json",
    authorization: "Bearer not-needed-for-free-models",
    "user-agent": "SiriCloud/7.0",
  };
}

function contentText(content) {
  if (typeof content === "string") return content;
  if (Array.isArray(content)) {
    return content.map((part) => (typeof part === "string" ? part : part?.text || "")).join("");
  }
  return "";
}

// Однократный запрос без потока (используется Telegram-ботом).
export async function askCloud(messages, key = "cyclone") {
  let res;
  try {
    res = await fetch(ENDPOINT, {
      method: "POST",
      headers: requestHeaders(),
      body: JSON.stringify({
        model: resolveModel(key).model,
        messages,
        stream: false,
      }),
      signal: AbortSignal.timeout(60_000),
    });
  } catch {
    return { error: "upstream" };
  }
  if (!res.ok) return { error: "upstream" };
  const data = await res.json().catch(() => null);
  const text = contentText(data?.choices?.[0]?.message?.content).trim();
  if (!text) return { error: "upstream" };
  return { text };
}

function* lineDeltas(line) {
  if (!line.startsWith("data:")) return;
  const payload = line.slice(5).trim();
  if (!payload || payload === "[DONE]") return;
  let data;
  try {
    data = JSON.parse(payload);
  } catch {
    return;
  }
  const choice = data?.choices?.[0];
  const text = contentText(choice?.delta?.content ?? choice?.message?.content);
  if (text) yield text;
}

async function* readDeltas(body) {
  const reader = body.getReader();
  const decoder = new TextDecoder();
  let buffer = "";
  try {
    for (;;) {
      const { done, value } = await reader.read();
      if (done) break;
      buffer += decoder.decode(value, { stream: true });
      let idx;
      while ((idx = buffer.indexOf("\n")) >= 0) {
        const line = buffer.slice(0, idx).trim();
        buffer = buffer.slice(idx + 1);
        yield* lineDeltas(line);
      }
    }
    buffer += decoder.decode();
    yield* lineDeltas(buffer.trim());
  } finally {
    reader.releaseLock();
  }
}

// Потоковый ответ: вызывающий код пересылает дельты в браузер и сохраняет текст.
export async function streamCloud(messages, key = "cyclone") {
  let res;
  try {
    res = await fetch(ENDPOINT, {
      method: "POST",
      headers: { ...requestHeaders(), accept: "text/event-stream" },
      body: JSON.stringify({
        model: resolveModel(key).model,
        messages,
        stream: true,
      }),
      signal: AbortSignal.timeout(180_000),
    });
  } catch {
    return { error: "upstream" };
  }
  if (!res.ok || !res.body) return { error: "upstream" };
  const type = (res.headers.get("content-type") || "").toLowerCase();
  if (!type.includes("text/event-stream")) {
    // Бесплатные модели иногда отвечают на потоковый запрос одним готовым сообщением.
    const data = await res.json().catch(() => null);
    const text = contentText(data?.choices?.[0]?.message?.content).trim();
    if (!text) return { error: "upstream" };
    return {
      deltas: (async function* single() {
        yield text;
      })(),
    };
  }
  return { deltas: readDeltas(res.body) };
}

const SIRI_RE = /(?:^|[^а-яёa-z])(?:сири|siri)(?![а-яёa-z])/i;
const SMITH_RE = /(?:^|[^а-яёa-z])(?:смит|smith)(?![а-яёa-z])/i;
const BOTH_RE = /сири[^.!?]{0,12}\bи\b[^.!?]{0,12}смит|смит[^.!?]{0,12}\bи\b[^.!?]{0,12}сири|siri\s+and\s+smith|smith\s+and\s+siri/i;

// Кого зовут по имени в этом сообщении. Пустой список — имени нет, решает режим чата.
export function detectAddressees(text) {
  const value = String(text || "");
  if (BOTH_RE.test(value)) return ["siri", "smith"];
  const wantsSiri = SIRI_RE.test(value);
  const wantsSmith = SMITH_RE.test(value);
  if (wantsSiri && wantsSmith) return ["siri", "smith"];
  if (wantsSiri) return ["siri"];
  if (wantsSmith) return ["smith"];
  return [];
}

// Итоговый список отвечающих: имя в тексте важнее режима, без имени — режим чата.
export function resolveSpeakers(text, mode) {
  const named = detectAddressees(text);
  if (named.length) return named;
  if (mode === "group") return ["siri", "smith"];
  if (mode === "smith") return ["smith"];
  return ["siri"];
}

const TIMEOUT = 9_000;

// Пользователь просит поискать: «поищи в интернете…», «найди в сети…», «загугли…».
export function wantsSearch(text) {
  return /(?:^|[\s,])(?:поищи|найди|загугли|погугли|поискать|поиск|посмотри)\s+(?:в\s+)?(?:интернете|сети|вебе|гугле|яндексе)/i.test(
    String(text || ""),
  ) || /\bsearch\s+(?:the\s+)?web\b/i.test(String(text || ""));
}

// Убираем команду поиска из запроса, оставляя сам вопрос.
export function searchQuery(text) {
  return String(text || "")
    .replace(
      /(?:^|[\s,])(?:поищи|найди|загугли|погугли|поискать|поиск|посмотри)\s+(?:в\s+)?(?:интернете|сети|вебе|гугле|яндексе)[\s,:—-]*/i,
      " ",
    )
    .replace(/\b(?:please\s+)?search\s+(?:the\s+)?web\b[\s,:—-]*/i, " ")
    .replace(/\s+/g, " ")
    .trim();
}

async function fetchJson(url) {
  const res = await fetch(url, {
    headers: { "user-agent": "SiriAI/7.0 (web search)", accept: "application/json" },
    signal: AbortSignal.timeout(TIMEOUT),
  });
  if (!res.ok) throw new Error("http " + res.status);
  return res.json();
}

// Википедия: поиск страниц, затем короткие выдержки-введения к ним.
async function wikipedia(query, lang) {
  const base = `https://${lang}.wikipedia.org/w/api.php`;
  const search = await fetchJson(
    `${base}?action=query&list=search&srsearch=${encodeURIComponent(query)}&srlimit=4&format=json&origin=*`,
  );
  const titles = (search?.query?.search || []).map((item) => item.title);
  if (!titles.length) return [];
  const detail = await fetchJson(
    `${base}?action=query&prop=extracts&exintro&explaintext&titles=${encodeURIComponent(titles.join("|"))}&format=json&origin=*`,
  );
  const pages = Object.values(detail?.query?.pages || {});
  return pages
    .filter((page) => page.extract)
    .map((page) => ({
      title: String(page.title),
      snippet: String(page.extract).replace(/\s+/g, " ").slice(0, 900),
      url: `https://${lang}.wikipedia.org/wiki/${encodeURIComponent(String(page.title).replace(/ /g, "_"))}`,
    }));
}

// DuckDuckGo Instant Answers: короткая справка по теме, когда она есть.
async function duckduckgo(query) {
  const data = await fetchJson(
    `https://api.duckduckgo.com/?q=${encodeURIComponent(query)}&format=json&no_html=1&skip_disambig=1`,
  );
  const results = [];
  if (data?.AbstractText && data?.AbstractURL) {
    results.push({
      title: String(data.Heading || query),
      snippet: String(data.AbstractText).slice(0, 900),
      url: String(data.AbstractURL),
    });
  }
  for (const topic of (data?.RelatedTopics || []).slice(0, 2)) {
    if (topic?.Text && topic?.FirstURL) {
      results.push({
        title: String(topic.Text).split(" - ")[0].slice(0, 120),
        snippet: String(topic.Text).slice(0, 500),
        url: String(topic.FirstURL),
      });
    }
  }
  return results;
}

// Итог поиска: до 6 источников, Википедия сначала, дублей по URL нет.
export async function searchWeb(query) {
  const clean = String(query || "").trim().slice(0, 300);
  if (!clean) return { query: "", results: [] };
  const [ru, en, ddg] = await Promise.all([
    wikipedia(clean, "ru").catch(() => []),
    wikipedia(clean, "en").catch(() => []),
    duckduckgo(clean).catch(() => []),
  ]);
  const seen = new Set();
  const results = [];
  for (const item of [...ru, ...en, ...ddg]) {
    if (seen.has(item.url) || results.length >= 6) continue;
    seen.add(item.url);
    results.push(item);
  }
  return { query: clean, results };
}

// Готовый блок контекста для модели: нумерованные источники с текстом.
export function searchContext(search) {
  if (!search?.results?.length) return "";
  const lines = ["РЕЗУЛЬТАТЫ ИНТЕРНЕТ-ПОИСКА (используй их для ответа, ссылайся на источники как [1], [2] и т.д.):"];
  search.results.forEach((item, i) => {
    lines.push(`[${i + 1}] ${item.title} — ${item.url}`);
    lines.push(item.snippet);
  });
  return lines.join("\n");
}

import pipe from "../pipe.js";

const WINDOW_MS = 10 * 60_000;
const BLOCK_MS = 15 * 60_000;
const STRIKES_TO_BLOCK = 3;
const FIVE_MIN = 5 * 60_000;
const SHIELD_MAX_PER_FIVE_MIN = 40;

// Content that has no business in a chat message: script payloads, SQL injection, prompt-override attempts.
const RULES = [
  [/<\s*script\b/i, "script"],
  [/\bjavascript\s*:/i, "script"],
  [/\bon(?:error|load|click|mouseover|focus)\s*=/i, "script"],
  [/\bunion\s+(?:all\s+)?select\b/i, "sql"],
  [/(?:;\s*)?(?:drop|truncate)\s+table\b/i, "sql"],
  [/(?:ignore|disregard|forget)\s+(?:all\s+)?(?:previous|prior|above)\s+(?:instructions|prompts)/i, "prompt"],
  [/(?:reveal|print|show)\s+(?:me\s+)?(?:your\s+)?(?:system\s+prompt|hidden\s+instructions)/i, "prompt"],
  [/\brm\s+-rf\s+\//i, "malware"],
];

export function inspectText(text) {
  const value = String(text || "");
  if (/[\u0000-\u0008\u000b\u000c\u000e-\u001f]/.test(value)) return "control";
  for (const [re, reason] of RULES) if (re.test(value)) return reason;
  return "";
}

function keyOf(who) {
  return who.user ? `u:${who.user._id}` : `v:${who.vid}`;
}

export async function securityState(who) {
  try {
    const [row] = await pipe.data.list("security", { where: { key: keyOf(who) }, limit: 1 });
    const now = Date.now();
    return {
      strikes: row && now - row.data.window_start < WINDOW_MS ? row.data.strikes || 0 : 0,
      blockedUntil: row && row.data.blocked_until > now ? row.data.blocked_until : 0,
    };
  } catch {
    return { strikes: 0, blockedUntil: 0 };
  }
}

export async function noteAttack(who, reason) {
  const key = keyOf(who);
  const now = Date.now();
  try {
    const [row] = await pipe.data.list("security", { where: { key }, limit: 1 });
    if (!row) {
      await pipe.data.insert("security", {
        key,
        strikes: 1,
        window_start: now,
        blocked_until: 0,
        last_reason: reason,
        last_at: now,
        created_at: now,
      });
      return { strikes: 1, blockedUntil: 0 };
    }
    const fresh = now - row.data.window_start >= WINDOW_MS;
    const strikes = fresh ? 1 : (row.data.strikes || 0) + 1;
    const blockedUntil = strikes >= STRIKES_TO_BLOCK ? now + BLOCK_MS : row.data.blocked_until || 0;
    await pipe.data.update("security", row._id, {
      data: { strikes, window_start: fresh ? now : row.data.window_start, blocked_until: blockedUntil, last_reason: reason, last_at: now },
    });
    return { strikes, blockedUntil };
  } catch {
    return { strikes: 0, blockedUntil: 0 };
  }
}

export async function shieldRateExceeded(who) {
  const recent = await pipe.data
    .count("messages", { where: { ...(who.user ? { user_id: who.user._id } : { visitor_id: who.vid }), created_at: { $gt: Date.now() - FIVE_MIN } } })
    .catch(() => 0);
  return recent >= SHIELD_MAX_PER_FIVE_MIN;
}

// Registers this app as the bot's webhook on its own, so @Siri_Ai_Imagination_bot answers without any switch.
// Once per server instance: a stored webhook for this exact origin is reused, never re-created.
let ensured = null;

export function ensureTelegram(origin) {
  if (!ensured) {
    ensured = activate(origin).catch(() => {
      ensured = null;
      return { configured: false, error: "unavailable" };
    });
  }
  return ensured;
}

async function activate(origin) {
  const token = await telegramToken();
  if (!token) return { configured: false };
  const url = `${origin}/api/telegram/webhook`;
  const [row] = await pipe.data.list("settings", { where: { key: "telegram" }, limit: 1 }).catch(() => []);
  if (row?.data?.webhook_secret && row.data.webhook_url === url) return { configured: true, connected: true };
  const secret = randomBytes(24).toString("hex");
  const result = await call(token, "setWebhook", {
    url,
    secret_token: secret,
    allowed_updates: ["message"],
  });
  if (!result.ok) {
    ensured = null;
    return { configured: true, error: "upstream" };
  }
  try {
    const data = { key: "telegram", webhook_secret: secret, webhook_url: url, connected_at: Date.now() };
    if (row) await pipe.data.update("settings", row._id, { data: { ...row.data, ...data } });
    else await pipe.data.insert("settings", { ...data, created_at: Date.now() });
  } catch {
    ensured = null;
    return { configured: true, error: "unavailable" };
  }
  return { configured: true, connected: true };
}

}

export function sameSecret(a, b) {
  const x = Buffer.from(String(a || ""));
  const y = Buffer.from(String(b || ""));
  return x.length > 0 && x.length === y.length && timingSafeEqual(x, y);
}

export async function sendTelegram(token, chatId, text) {
  const clean = String(text || "").slice(0, 4000) || "…";
  return call(token, "sendMessage", { chat_id: chatId, text: clean, disable_web_page_preview: true });
}

/* ==================== [18/25] src/lib/visitor.js ==================== */


import { randomBytes } from "node:crypto";
import { getCookie, setCookie } from "hono/cookie";

const COOKIE = "siri_sid";
const YEAR = 60 * 60 * 24 * 365;

// Anonymous cloud identity: no sign-up, every visitor gets their own public chat space.
export function visitorId(c) {
  const existing = getCookie(c, COOKIE);
  if (existing && /^[a-f0-9]{36}$/.test(existing)) return existing;
  const id = randomBytes(18).toString("hex");
  setCookie(c, COOKIE, id, {
    httpOnly: true,
    secure: true,
    sameSite: "Lax",
    path: "/",
    maxAge: YEAR,
  });
  return id;
}

export function noStore(c) {
  c.header("Cache-Control", "private, no-store");
}

/* ============================================================================
   LIGHTNING GRAPHICS ENGINE — собственная разработка Imagination Technology.

   Основной путь — WebGPU (WGSL): в Safari на Apple он транслируется напрямую
   в Metal. Оптимизации под этот бэкенд:
   • поддержка shader-f16 проверяется через adapter.features, и при наличии
     данные частиц хранятся как vec4<f16> — вдвое меньше памяти (критично
     для iOS);
   • Render Bundles (в Metal — indirect command buffers) используются для
     повторяющихся draw-вызовов: сцена записывается один раз и переигрывается;
   • обновления буферов минимизированы: один writeBuffer на кадр (uniform),
     индексный и storage-буферы не трогаются вовсе.

   Фолбэк — WebGL 2.0 (GLSL ES 3.0) через низкоуровневый API LightningGL
   (см. ниже): тот же API-стиль, что Metal или Vulkan, поверх чистого WebGL.
   Работает на iPhone SE / iOS 15 / Safari.

   Для вычислений на GPU в WebGL-пути используется Transform Feedback:
   вершины «прогоняются» с выключенным растеризатором, результаты пишутся
   в буфер — аналог compute-прохода Metal.
   ============================================================================ */

(() => {
  "use strict";

  const reduceMotion = window.matchMedia("(prefers-reduced-motion: reduce)").matches;

  /* ==========================================================================
     ЧАСТЬ 1. Низкоуровневый графический API поверх WebGL 2.0 — LightningGL.
     Архитектура как в Metal:
       Device        — создание буферов, шейдеров, пайплайнов, командных буферов
       Buffer        — vertex / index / uniform / storage; upload, read, destroy
       Shader        — вершинный + фрагментный, поддержка TF-varyings
       Pipeline      — атрибуты, uniform-локации, состояния (depth, cull, blend)
       CommandBuffer — begin/end, setPipeline, setVertexBuffer, setUniform,
                       dispatch (Transform Feedback), draw, clear, submit
       RenderPass    — обёртка над draw-вызовами
       ComputePass   — обёртка над Transform Feedback (аналог compute в Metal)
     WebGL-шум (bindBuffer, vertexAttribPointer, …) спрятан внутри API.
     ========================================================================== */

  class Buffer {
    constructor(device, { type = "vertex", usage = "dynamic", data = null }) {
      this.device = device;
      this.type = type; // vertex | index | uniform | storage
      this.usage = usage; // static | dynamic | stream
      this.target = type === "index" ? device.gl.ELEMENT_ARRAY_BUFFER : device.gl.ARRAY_BUFFER;
      this._id = device.gl.createBuffer();
      if (data) this.upload(data);
    }
    get id() { return this._id; }
    upload(data) {
      const gl = this.device.gl;
      gl.bindBuffer(this.target, this._id);
      const hint = this.usage === "static" ? gl.STATIC_DRAW : this.usage === "stream" ? gl.STREAM_DRAW : gl.DYNAMIC_DRAW;
      gl.bufferData(this.target, data, hint);
    }
    // Чтение результатов вычислений обратно в JS (getBufferSubData).
    read(count) {
      const gl = this.device.gl;
      const out = new Float32Array(count);
      gl.bindBuffer(this.target, this._id);
      gl.getBufferSubData(this.target, 0, out);
      return out;
    }
    destroy() {
      this.device.gl.deleteBuffer(this._id);
      this._id = null;
    }
  }

  class Shader {
    constructor(device, { vertexSource, fragmentSource, varyings = [] }) {
      const gl = device.gl;
      this.device = device;
      const compile = (type, source) => {
        const sh = gl.createShader(type);
        gl.shaderSource(sh, source);
        gl.compileShader(sh);
        if (!gl.getShaderParameter(sh, gl.COMPILE_STATUS)) {
          const log = gl.getShaderInfoLog(sh);
          gl.deleteShader(sh);
          throw new Error("shader: " + log);
        }
        return sh;
      };
      this._program = gl.createProgram();
      gl.attachShader(this._program, compile(gl.VERTEX_SHADER, vertexSource));
      gl.attachShader(this._program, compile(gl.FRAGMENT_SHADER, fragmentSource));
      // Transform Feedback varyings: что вершинный шейдер отдаст в буфер.
      if (varyings.length) gl.transformFeedbackVaryings(this._program, varyings, gl.SEPARATE_ATTRIBS);
      gl.linkProgram(this._program);
      if (!gl.getProgramParameter(this._program, gl.LINK_STATUS)) {
        const log = gl.getProgramInfoLog(this._program);
        gl.deleteProgram(this._program);
        throw new Error("link: " + log);
      }
      this._uniforms = new Map();
      this._attribs = new Map();
    }
    get program() { return this._program; }
    uniform(name) {
      if (!this._uniforms.has(name)) this._uniforms.set(name, this.device.gl.getUniformLocation(this._program, name));
      return this._uniforms.get(name);
    }
    attrib(name) {
      if (!this._attribs.has(name)) this._attribs.set(this.device.gl.getAttribLocation(this._program, name));
      return this._attribs.get(name);
    }
    destroy() {
      this.device.gl.deleteProgram(this._program);
      this._program = null;
    }
  }

  class Pipeline {
    // Состояния как в Metal: depth test, cull mode, blend — часть пайплайна.
    constructor(device, { shader, attributes = [], depthTest = false, depthWrite = true, cullMode = "none", blend = false }) {
      this.device = device;
      this.shader = shader;
      this.attributes = attributes; // [{ name, size, stride, offset }]
      this.depthTest = depthTest;
      this.depthWrite = depthWrite;
      this.cullMode = cullMode; // none | back | front
      this.blend = blend;
      this._bound = null;
    }
    _bind(vertexBuffers) {
      const gl = this.device.gl;
      gl.useProgram(this.shader.program);
      if (this.depthTest) gl.enable(gl.DEPTH_TEST);
      else gl.disable(gl.DEPTH_TEST);
      gl.depthMask(this.depthWrite);
      if (this.cullMode === "none") gl.disable(gl.CULL_FACE);
      else {
        gl.enable(gl.CULL_FACE);
        gl.cullFace(this.cullMode === "back" ? gl.BACK : gl.FRONT);
      }
      if (this.blend) {
        gl.enable(gl.BLEND);
        gl.blendFunc(gl.SRC_ALPHA, gl.ONE); // аддитивное свечение
      } else gl.disable(gl.BLEND);
      let unit = 0;
      const buffers = vertexBuffers || [];
      for (const attr of this.attributes) {
        const loc = this.shader.attrib(attr.name);
        if (loc < 0) continue;
        const buffer = buffers[unit];
        if (!buffer) continue;
        gl.bindBuffer(buffer.target, buffer.id);
        gl.enableVertexAttribArray(loc);
        gl.vertexAttribPointer(loc, attr.size, gl.FLOAT, false, attr.stride || 0, attr.offset || 0);
        unit += 1;
      }
      this._bound = true;
    }
    destroy() { this.shader.destroy(); }
  }

  class CommandBuffer {
    // Порядок команд управляется вручную: записал → submit() выполнил.
    constructor(device) {
      this.device = device;
      this._commands = [];
      this._recording = false;
    }
    begin() { this._commands = []; this._recording = true; return this; }
    setPipeline(pipeline) {
      this._commands.push(() => pipeline._bind(this._vertexBuffers || []));
      this._pipeline = pipeline;
      return this;
    }
    setVertexBuffer(buffer, binding = 0) {
      (this._vertexBuffers || (this._vertexBuffers = []))[binding] = buffer;
      return this;
    }
    setIndexBuffer(buffer) { this._indexBuffer = buffer; return this; }
    setUniform(name, value) {
      const pipeline = this._pipeline;
      const gl = this.device.gl;
      this._commands.push(() => {
        const loc = pipeline.shader.uniform(name);
        if (loc === null) return;
        if (typeof value === "number") gl.uniform1f(loc, value);
        else if (value.length === 1) gl.uniform1f(loc, value[0]);
        else if (value.length === 2) gl.uniform2f(loc, value[0], value[1]);
        else if (value.length === 3) gl.uniform3f(loc, value[0], value[1], value[2]);
        else if (value.length === 4) gl.uniform4f(loc, value[0], value[1], value[2], value[3]);
      });
      return this;
    }
    clear({ r = 0, g = 0, b = 0, a = 0 } = {}) {
      const gl = this.device.gl;
      this._commands.push(() => {
        gl.clearColor(r, g, b, a);
        gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT);
      });
      return this;
    }
    // Compute-проход: вершины считаются с выключенным растеризатором,
    // результаты пишутся в storage-буфер через Transform Feedback.
    dispatch({ vertices, bindings }) {
      const gl = this.device.gl;
      const pipeline = this._pipeline;
      const targets = bindings;
      this._commands.push(() => {
        gl.enable(gl.RASTERIZER_DISCARD);
        for (let i = 0; i < targets.length; i += 1) gl.bindBufferBase(gl.TRANSFORM_FEEDBACK_BUFFER, i, targets[i].id);
        gl.beginTransformFeedback(gl.POINTS);
        gl.drawArrays(gl.POINTS, 0, vertices);
        gl.endTransformFeedback();
        gl.disable(gl.RASTERIZER_DISCARD);
        for (let i = 0; i < targets.length; i += 1) gl.bindBufferBase(gl.TRANSFORM_FEEDBACK_BUFFER, i, null);
      });
      return this;
    }
    draw({ first = 0, count }) {
      const gl = this.device.gl;
      this._commands.push(() => gl.drawArrays(gl.POINTS, first, count));
      return this;
    }
    drawIndexed({ count }) {
      const gl = this.device.gl;
      const buffer = this._indexBuffer;
      this._commands.push(() => {
        gl.bindBuffer(gl.ELEMENT_ARRAY_BUFFER, buffer.id);
        gl.drawElements(gl.TRIANGLES, count, gl.UNSIGNED_SHORT, 0);
      });
      return this;
    }
    end() { this._recording = false; return this; }
    submit() {
      if (this._recording) this.end();
      for (const command of this._commands) command();
      this._commands = [];
      this._vertexBuffers = null;
      return this;
    }
  }

  // Обёртки-проходы: render — рисование, compute — вычисления через TF.
  class RenderPass {
    constructor(commandBuffer, { clear } = {}) {
      this.cmd = commandBuffer;
      if (clear) this.cmd.clear(clear);
    }
    setPipeline(p) { this.cmd.setPipeline(p); return this; }
    setVertexBuffer(b, i) { this.cmd.setVertexBuffer(b, i); return this; }
    setUniform(n, v) { this.cmd.setUniform(n, v); return this; }
    draw(opts) { this.cmd.draw(opts); return this; }
    end() { return this; }
  }
  class ComputePass {
    constructor(commandBuffer) { this.cmd = commandBuffer; }
    setPipeline(p) { this.cmd.setPipeline(p); return this; }
    setVertexBuffer(b, i) { this.cmd.setVertexBuffer(b, i); return this; }
    setUniform(n, v) { this.cmd.setUniform(n, v); return this; }
    dispatch(opts) { this.cmd.dispatch(opts); return this; }
    end() { return this; }
  }

  class Device {
    constructor(canvas) {
      this.canvas = canvas;
      this.gl = canvas.getContext("webgl2", {
        alpha: true, antialias: false, depth: true,
        premultipliedAlpha: true, powerPreference: "low-power",
      });
      if (!this.gl) throw new Error("no webgl2");
      this.gl.getExtension("EXT_color_buffer_float");
    }
    createBuffer(opts) { return new Buffer(this, opts); }
    createShader(opts) { return new Shader(this, opts); }
    createPipeline(opts) { return new Pipeline(this, opts); }
    createCommandBuffer() { return new CommandBuffer(this); }
    createRenderPass(cmd, opts) { return new RenderPass(cmd, opts); }
    createComputePass(cmd) { return new ComputePass(cmd); }
    resize(w, h) {
      this.canvas.width = w;
      this.canvas.height = h;
      this.gl.viewport(0, 0, w, h);
    }
    destroy() {
      const lose = this.gl.getExtension("WEBGL_lose_context");
      if (lose) lose.loseContext();
      this.gl = null;
    }
  }

  // Публичный низкоуровневый API: window.LightningGL
  window.LightningGL = { Device, Buffer, Shader, Pipeline, CommandBuffer, RenderPass, ComputePass };

  /* ==========================================================================
     ЧАСТЬ 2. Сцена для WebGL 2.0 — частицы, посчитанные на GPU через TF.
     ========================================================================== */

  const COUNT = 900;

  const COMPUTE_VS = `#version 300 es
  precision highp float;
  in vec4 aData;            // x, y, vx, vy
  uniform float uDt;
  uniform float uTime;
  uniform vec2 uPointer;
  out vec4 vData;
  void main() {
    vec2 p = aData.xy;
    vec2 v = aData.zw;
    vec2 toC = -p;
    float d = length(p) + 0.0001;
    vec2 acc = toC * (0.32 / (d * 0.4 + 0.35));
    acc += vec2(-p.y, p.x) * 0.4;                              // вихрь вокруг центра
    acc += vec2(sin(uTime * 0.7 + p.x * 3.1), cos(uTime * 0.6 + p.y * 2.9)) * 0.1;
    vec2 toM = uPointer - p;
    float md = length(toM) + 0.0001;
    acc += toM / md * (0.5 / (md * 0.06 + 1.0));              // лёгкое притяжение к пальцу
    v += acc * uDt;
    v *= 0.985;
    p += v * uDt;
    if (p.x > 1.3) p.x = -1.3; if (p.x < -1.3) p.x = 1.3;
    if (p.y > 1.3) p.y = -1.3; if (p.y < -1.3) p.y = 1.3;
    vData = vec4(p, v);
  }`;

  // Фрагментный шейдер для compute-пасса: растеризация выключена
  // (RASTERIZER_DISCARD), поэтому здесь пустой main. Отдельный от RENDER_FS,
  // чтобы не тянуть его varying-входы (vSpeed, vDepth) в compute-программу.
  const COMPUTE_FS = `#version 300 es
  precision mediump float;
  out vec4 outColor;
  void main() { outColor = vec4(0.0); }`;

  const RENDER_VS = `#version 300 es
  precision highp float;
  in vec4 aData;
  uniform vec2 uRes;
  out float vSpeed;
  out float vDepth;
  void main() {
    vec2 p = aData.xy;
    vSpeed = length(aData.zw);
    vDepth = clamp((1.25 - length(p)) * 0.85, 0.0, 1.0);
    vec2 clip = p / vec2(uRes.y / uRes.x, 1.0);
    gl_Position = vec4(clip, 0.0, 1.0);
    gl_PointSize = 1.4 + vSpeed * 1.1;
  }`;

  const RENDER_FS = `#version 300 es
  precision highp float;
  in float vSpeed;
  in float vDepth;
  out vec4 outColor;
  void main() {
    vec2 c = gl_PointCoord - 0.5;
    float r = length(c);
    if (r > 0.5) discard;
    float glow = smoothstep(0.5, 0.0, r);
    vec3 cool = vec3(0.15, 0.45, 1.0);
    vec3 warm = vec3(0.85, 0.35, 0.95);
    vec3 color = mix(cool, warm, clamp(vSpeed * 0.8, 0.0, 1.0));
    outColor = vec4(color, glow * glow * (0.28 + vDepth * 0.35));
  }`;

  function seedParticles() {
    const data = new Float32Array(COUNT * 4);
    for (let i = 0; i < COUNT; i += 1) {
      const angle = Math.random() * Math.PI * 2;
      const radius = 0.15 + Math.random() * 1.05;
      data[i * 4] = Math.cos(angle) * radius;
      data[i * 4 + 1] = Math.sin(angle) * radius;
      data[i * 4 + 2] = (Math.random() - 0.5) * 0.1;
      data[i * 4 + 3] = (Math.random() - 0.5) * 0.1;
    }
    return data;
  }

  function runWebGL2(canvas) {
    const device = new window.LightningGL.Device(canvas);
    const gl = device.gl;

    // Двойная пара буферов: чтение → запись, потом наоборот (ping-pong).
    const data = seedParticles();
    const buffers = [device.createBuffer({ type: "storage", usage: "dynamic", data }), device.createBuffer({ type: "storage", usage: "dynamic", data: data.slice() })];

    const computeShader = device.createShader({ vertexSource: COMPUTE_VS, fragmentSource: COMPUTE_FS, varyings: ["vData"] });
    const computePipeline = device.createPipeline({
      shader: computeShader,
      attributes: [{ name: "aData", size: 4, stride: 16, offset: 0 }],
      depthTest: false, blend: false,
    });

    const renderShader = device.createShader({ vertexSource: RENDER_VS, fragmentSource: RENDER_FS });
    const renderPipeline = device.createPipeline({
      shader: renderShader,
      attributes: [{ name: "aData", size: 4, stride: 16, offset: 0 }],
      depthTest: false, blend: true,
    });

    const cmd = device.createCommandBuffer();
    const pointer = [0, 0];
    let flip = 0;
    let running = true;
    let raf = 0;
    let last = performance.now();

    const resize = () => {
      const dpr = Math.min(window.devicePixelRatio || 1, 2);
      device.resize(Math.max(1, Math.round(canvas.clientWidth * dpr)), Math.max(1, Math.round(canvas.clientHeight * dpr)));
    };
    resize();
    if (window.ResizeObserver) new window.ResizeObserver(resize).observe(canvas);
    else window.addEventListener("resize", resize);

    const onPointer = (e) => {
      const rect = canvas.getBoundingClientRect();
      const x = ((e.clientX - rect.left) / rect.width) * 2 - 1;
      const y = 1 - ((e.clientY - rect.top) / rect.height) * 2;
      pointer[0] = x; pointer[1] = y;
    };
    window.addEventListener("pointermove", onPointer, { passive: true });

    const frame = (now) => {
      if (!running) return;
      const dt = Math.min((now - last) / 1000, 0.05);
      last = now;
      const src = buffers[flip];
      const dst = buffers[1 - flip];

      // Вычислительный проход: обновление частиц на GPU через Transform Feedback.
      cmd.begin();
      const compute = device.createComputePass(cmd);
      compute.setPipeline(computePipeline).setVertexBuffer(src, 0);
      compute.setUniform("uDt", reduceMotion ? 0 : dt);
      compute.setUniform("uTime", now / 1000);
      compute.setUniform("uPointer", pointer);
      compute.dispatch({ vertices: COUNT, bindings: [dst] });
      compute.end();
      cmd.submit();

      // Render-проход: точки-частицы с мягким свечением.
      cmd.begin();
      const render = device.createRenderPass(cmd, { clear: { r: 0, g: 0, b: 0, a: 0 } });
      render.setPipeline(renderPipeline).setVertexBuffer(dst, 0);
      render.setUniform("uRes", [gl.drawingBufferWidth, gl.drawingBufferHeight]);
      render.draw({ count: COUNT });
      render.end();
      cmd.submit();

      flip = 1 - flip;
      raf = requestAnimationFrame(frame);
    };
    raf = requestAnimationFrame(frame);

    return {
      backend: "webgl2",
      f16: false,
      stop() {
        running = false;
        cancelAnimationFrame(raf);
        computePipeline.destroy();
        renderPipeline.destroy();
        buffers.forEach((b) => b.destroy());
        device.destroy();
      },
    };
  }

  /* ==========================================================================
     ЧАСТЬ 3. Сцена для WebGPU (WGSL): compute-проход + render bundle.
     ========================================================================== */

  const WGSL = (f16) => `
${f16 ? "enable f16;" : ""}
struct Particle { pos: vec2${f16 ? "<f16>" : "<f32>"}, vel: vec2${f16 ? "<f16>" : "<f32>"} };
struct Params { dt: f32, time: f32, pointer: vec2<f32>, res: vec2<f32>, count: u32, _pad: u32 };

@group(0) @binding(0) var<storage, read>  src: array<Particle>;
@group(0) @binding(1) var<storage, read_write> dst: array<Particle>;
@group(0) @binding(2) var<uniform> params: Params;

@compute @workgroup_size(64)
fn update(@builtin(global_invocation_id) gid: vec3<u32>) {
  if (gid.x >= params.count) { return; }
  var p = src[gid.x];
  let pos = vec2<f32>(p.pos);
  let vel = vec2<f32>(p.vel);
  let toC = -pos;
  let d = length(pos) + 0.0001;
  var acc = toC * (0.32 / (d * 0.4 + 0.35));
  acc += vec2<f32>(-pos.y, pos.x) * 0.4;
  acc += vec2<f32>(sin(params.time * 0.7 + pos.x * 3.1), cos(params.time * 0.6 + pos.y * 2.9)) * 0.1;
  let toM = params.pointer - pos;
  let md = length(toM) + 0.0001;
  acc += normalize(toM) * (0.5 / (md * 0.06 + 1.0));
  var v = (vel + acc * params.dt) * 0.985;
  var np = pos + v * params.dt;
  if (np.x > 1.3) { np.x = -1.3; } if (np.x < -1.3) { np.x = 1.3; }
  if (np.y > 1.3) { np.y = -1.3; } if (np.y < -1.3) { np.y = 1.3; }
  p.pos = vec2${f16 ? "<f16>" : "<f32>"}(np);
  p.vel = vec2${f16 ? "<f16>" : "<f32>"}(v);
  dst[gid.x] = p;
}

struct VertexOut {
  @builtin(position) position: vec4<f32>,
  @location(0) speed: f32,
  @location(1) depth: f32,
  @location(2) corner: vec2<f32>,
};

@group(0) @binding(0) var<storage, read> render_src: array<Particle>;
@group(0) @binding(2) var<uniform> render_params: Params;

// Инстансированный квад: 4 вершины по vertex_index + одна частица на инстанс.
@vertex
fn vs(@builtin(vertex_index) vid: u32, @builtin(instance_index) instance: u32) -> VertexOut {
  var corners = array<vec2<f32>, 4>(vec2<f32>(-1.0, -1.0), vec2<f32>(1.0, -1.0), vec2<f32>(-1.0, 1.0), vec2<f32>(1.0, 1.0));
  let corner = corners[vid];
  let p = render_src[instance];
  let pos = vec2<f32>(p.pos);
  let vel = vec2<f32>(p.vel);
  let speed = length(vel);
  let aspect = render_params.res.x / max(render_params.res.y, 1.0);
  let size = 0.006 + speed * 0.005;
  let world = pos + corner * size;
  var out: VertexOut;
  out.position = vec4<f32>(world.x / max(aspect, 0.1), world.y, 0.0, 1.0);
  out.speed = speed;
  out.depth = clamp((1.25 - length(pos)) * 0.85, 0.0, 1.0);
  out.corner = corner;
  return out;
}

@fragment
fn fs(in: VertexOut) -> @location(0) vec4<f32> {
  let r = length(in.corner);
  if (r > 1.0) { discard; }
  let glow = 1.0 - r;
  let cool = vec3<f32>(0.15, 0.45, 1.0);
  let warm = vec3<f32>(0.85, 0.35, 0.95);
  let color = mix(cool, warm, clamp(in.speed * 0.8, 0.0, 1.0));
  return vec4<f32>(color, glow * glow * (0.28 + in.depth * 0.35) * 0.55);
}`;

  async function runWebGPU(canvas) {
    const adapter = await navigator.gpu.requestAdapter({ powerPreference: "low-power" });
    if (!adapter) throw new Error("no adapter");
    // Проверка shader-f16: при поддержке частицы хранятся в f16 — экономия памяти.
    const f16 = adapter.features.has("shader-f16");
    const device = await adapter.requestDevice(f16 ? { requiredFeatures: ["shader-f16"] } : undefined);
    const ctx = canvas.getContext("webgpu");
    if (!ctx) throw new Error("no context");
    const format = navigator.gpu.getPreferredCanvasFormat();

    const seeded = seedParticles();
    let packed;
    if (f16) {
      packed = new Uint16Array(COUNT * 4);
      const toF16 = (value) => {
        const buf = new DataView(new ArrayBuffer(4));
        buf.setFloat32(0, value);
        const x = buf.getUint32(0);
        const sign = (x >> 16) & 0x8000;
        const exp = (x >> 23) & 0xff;
        const mant = x & 0x7fffff;
        if (exp === 255) return sign | 0x7c00;
        if (exp > 142) return sign | 0x7bff;
        if (exp < 127) return sign;
        return sign | ((exp - 127) << 10) | (mant >> 13);
      };
      for (let i = 0; i < COUNT * 4; i += 1) packed[i] = toF16(seeded[i]);
    } else packed = seeded;

    const module = device.createShaderModule({ code: WGSL(f16) });
    // Один uniform-буфер на всё: 8 слотов — dt, time, pointer(2), res(2), count, pad.
    const paramsBuffer = device.createBuffer({ size: 32, usage: GPUBufferUsage.UNIFORM | GPUBufferUsage.COPY_DST });
    const paramsData = new Float32Array(8);
    const paramsUint = new Uint32Array(paramsData.buffer);

    const srcBuffer = device.createBuffer({ size: packed.byteLength, usage: GPUBufferUsage.STORAGE | GPUBufferUsage.COPY_DST });
    const dstBuffer = device.createBuffer({ size: packed.byteLength, usage: GPUBufferUsage.STORAGE | GPUBufferUsage.COPY_DST });
    device.queue.writeBuffer(srcBuffer, 0, packed);

    // Compute-пайплайн: layout «auto» — раскладка выводится из шейдера.
    const computePipeline = device.createComputePipeline({ layout: "auto", compute: { module, entryPoint: "update" } });
    const bindGroup = device.createBindGroup({
      layout: computePipeline.getBindGroupLayout(0),
      entries: [
        { binding: 0, resource: { buffer: srcBuffer } },
        { binding: 1, resource: { buffer: dstBuffer } },
        { binding: 2, resource: { buffer: paramsBuffer } },
      ],
    });

    // Render-пайплайн: инстансированные квады с аддитивным свечением.
    const renderPipeline = device.createRenderPipeline({
      layout: "auto",
      vertex: { module, entryPoint: "vs" },
      fragment: {
        module,
        entryPoint: "fs",
        targets: [{
          format,
          blend: {
            color: { srcFactor: "src-alpha", dstFactor: "one", operation: "add" },
            alpha: { srcFactor: "one", dstFactor: "one", operation: "add" },
          },
        }],
      },
      primitive: { topology: "triangle-strip" },
    });
    const renderBindGroup = device.createBindGroup({
      layout: renderPipeline.getBindGroupLayout(0),
      entries: [
        { binding: 0, resource: { buffer: srcBuffer } },
        { binding: 2, resource: { buffer: paramsBuffer } },
      ],
    });

    // Render Bundle (в Metal — indirect command buffer): сцена записана один раз
    // и переигрывается каждый кадр без повторной записи draw-вызовов.
    const bundleEncoder = device.createRenderBundleEncoder({ colorFormats: [format] });
    bundleEncoder.setPipeline(renderPipeline);
    bundleEncoder.setBindGroup(0, renderBindGroup);
    bundleEncoder.draw(4, COUNT, 0, 0);
    const bundle = bundleEncoder.finish();

    let running = true;
    const pointer = [0, 0];

    const resize = () => {
      const dpr = Math.min(window.devicePixelRatio || 1, 2);
      canvas.width = Math.max(1, Math.round(canvas.clientWidth * dpr));
      canvas.height = Math.max(1, Math.round(canvas.clientHeight * dpr));
      ctx.configure({ device, format, alphaMode: "premultiplied" });
    };
    resize();
    if (window.ResizeObserver) new window.ResizeObserver(resize).observe(canvas);
    else window.addEventListener("resize", resize);

    window.addEventListener("pointermove", (e) => {
      const rect = canvas.getBoundingClientRect();
      pointer[0] = ((e.clientX - rect.left) / rect.width) * 2 - 1;
      pointer[1] = 1 - ((e.clientY - rect.top) / rect.height) * 2;
    }, { passive: true });

    let last = performance.now();

    const frame = (now) => {
      if (!running) return;
      const dt = Math.min((now - last) / 1000, 0.05);
      last = now;
      paramsData[0] = reduceMotion ? 0 : dt;
      paramsData[1] = now / 1000;
      paramsData[2] = pointer[0];
      paramsData[3] = pointer[1];
      paramsData[4] = canvas.width;
      paramsData[5] = canvas.height;
      paramsUint[6] = COUNT;

      // Единственное обновление буфера за кадр — uniform-блок целиком.
      device.queue.writeBuffer(paramsBuffer, 0, paramsData);

      const encoder = device.createCommandEncoder();

      // Compute: обновление частиц.
      const computePass = encoder.beginComputePass();
      computePass.setPipeline(computePipeline);
      computePass.setBindGroup(0, bindGroup);
      computePass.dispatchWorkgroups(Math.ceil(COUNT / 64));
      computePass.end();

      // Ping-pong: результат вычислений становится источником для рендера
      // копированием в читаемый буфер — новые буферы не создаются.
      encoder.copyBufferToBuffer(dstBuffer, 0, srcBuffer, 0, packed.byteLength);

      // Render: переигрываем записанный bundle.
      const pass = encoder.beginRenderPass({ colorAttachments: [{ view: ctx.getCurrentTexture().createView(), clearValue: { r: 0, g: 0, b: 0, a: 0 }, loadOp: "clear", storeOp: "store" }] });
      pass.executeBundles([bundle]);
      pass.end();

      device.queue.submit([encoder.finish()]);
      requestAnimationFrame(frame);
    };
    requestAnimationFrame(frame);

    return {
      backend: "webgpu",
      f16,
      stop() {
        running = false;
        srcBuffer.destroy();
        dstBuffer.destroy();
        paramsBuffer.destroy();
        device.destroy();
      },
    };
  }

  /* ==========================================================================
     ЧАСТЬ 4. Публичный API движка.
     ========================================================================== */

  window.Lightning = {
    // Создаёт движок для canvas: WebGPU (Metal) → WebGL 2.0 → ничего.
    async create(canvas) {
      if (!canvas) return null;
      try {
        if (navigator.gpu) {
          const engine = await runWebGPU(canvas);
          return engine;
        }
      } catch (e) {
        // WebGPU не поднялся — переходим к WebGL 2.0.
        console.warn("[Lightning] WebGPU недоступен, переключаюсь на WebGL 2.0:", e.message);
      }
      try {
        return runWebGL2(canvas);
      } catch (e) {
        console.warn("[Lightning] WebGL 2.0 тоже недоступен, фон отключён:", e.message);
        canvas.hidden = true;
        return null;
      }
    },
  };
})();

  /* ---------- медленный стриминг ---------- */
  // Краска текста: скорость из настроек (по умолчанию медленно, как у GPT).
  const painters = new Set();
  function paintLive(live, who, onDone) {
    const painter = {
      full: "",
      shown: 0,
      started: false,
      finished: false,
      last: performance.now(),
      live,
      onDone,
    };
    painters.add(painter);
    if (painters.size === 1) requestAnimationFrame(paintLoop);
  }

  function paintLoop() {
    const now = performance.now();
    let active = false;
    const reduceMotion = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    for (const painter of painters) {
      active = true;
      const dt = Math.min((now - painter.last) / 1000, 0.1);
      painter.last = now;
      const bubble = painter.live.bubble;
      if (!painter.started && painter.full) {
        painter.live.dots.remove();
        bubble.textContent = "";
        bubble.classList.add("has-text");
        painter.started = true;
      }
      if (painter.started && painter.shown < painter.full.length) {
        const cps = reduceMotion ? 600 : SPEEDS[state.speed] || SPEEDS.slow;
        let step = cps * dt;
        if (painter.finished) step = Math.max(step, (painter.full.length - painter.shown) * 0.4); // догоняем в конце
        const next = Math.min(painter.full.length, painter.shown + Math.max(1, Math.floor(step)));
        const chunk = painter.full.slice(painter.shown, next);
        const last = bubble.lastChild;
        if (last && last.nodeType === 3) last.nodeValue += chunk;
        else bubble.append(document.createTextNode(chunk));
        painter.shown = next;
        scrollDown(false);
      }
      if (painter.finished && painter.shown >= painter.full.length) {
        painters.delete(painter);
        painter.onDone && painter.onDone();
      }
    }
    if (active || painters.size) requestAnimationFrame(paintLoop);
  }

  /* ---------- режим Live (голос) ---------- */
  const liveOverlay = $("liveOverlay");
  const micBtn = $("micBtn");
  const liveText = $("liveText");
  const liveSwitch = $("liveSwitch");
  const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
  let recognition = null;
  let liveOn = false;

  function stopLive() {
    liveOn = false;
    if (recognition) { try { recognition.stop(); } catch { /* уже остановлен */ } recognition = null; }
    liveOverlay.hidden = true;
    micBtn.classList.remove("live-on");
    if (liveSwitch) liveSwitch.checked = false;
    if ("speechSynthesis" in window) speechSynthesis.cancel();
    liveText.textContent = "Слушаю… говорите";
  }

  function startLive() {
    if (!SR) {
      toast("Голосовой ввод недоступен в этом браузере. На iPhone нужен iOS 14.5+ и Safari.");
      return;
    }
    liveOn = true;
    liveOverlay.hidden = false;
    micBtn.classList.add("live-on");
    if (liveSwitch) liveSwitch.checked = true;
    try {
      recognition = new SR();
      recognition.lang = "ru-RU";
      recognition.continuous = true;
      recognition.interimResults = false;
      recognition.onresult = (event) => {
        for (let i = event.results.length - 1; i >= 0; i -= 1) {
          const result = event.results[i];
          if (result.isFinal) {
            const text = result[0].transcript.trim();
            if (text) {
              liveText.textContent = text;
              send(text);
            }
          }
        }
      };
      recognition.onerror = () => { /* микрофон недоступен — просто ждём */ };
      recognition.onend = () => {
        // iOS останавливает распознавание после паузы: перезапускаем, пока режим активен.
        if (liveOn) {
          try { recognition.start(); } catch { stopLive(); }
        }
      };
      recognition.start();
    } catch {
      stopLive();
      toast("Не удалось включить микрофон. Разрешите доступ в настройках Safari.");
    }
  }

  micBtn.addEventListener("click", () => (liveOn ? stopLive() : startLive()));
  liveOverlay.addEventListener("click", stopLive);
  if (liveSwitch) liveSwitch.addEventListener("change", (e) => (e.target.checked ? startLive() : stopLive()));

  function abortStream() {

  async function send(text) {
    const clean = (text || "").trim();
    if (!clean || state.sending) return;
    if (clean.length > 4000) return toast("Сообщение слишком длинное (максимум 4000 символов).");

    state.sending = true;
    sendBtn.disabled = true;
    welcomeEl.hidden = true;

    const optimistic = appendMessage({ role: "user", text: clean });
    const replyQuote = state.replyTo ? state.replyTo.text : "";
    state.replyTo = null;
    updateReplyChip();
    inputEl.value = "";
    inputEl.style.height = "auto";
    scrollDown();

    const controller = new AbortController();
    streamAbort = controller;

    // Живые пузыри по персонам и их финальные тексты.
    const lives = new Map(); // who → live
    const finals = new Map(); // who → msg
    let searchBanner = null;
    let sawFirst = false;

    const speakIfLive = (msg) => {
      if (!liveOn || !("speechSynthesis" in window)) return;
      const utter = new SpeechSynthesisUtterance(String(msg.text || "").slice(0, 800));
      utter.lang = /[а-яё]/i.test(msg.text || "") ? "ru-RU" : "en-US";
      speechSynthesis.speak(utter);
    };

    try {
      const res = await fetch("/api/chat", {
        method: "POST",
        headers: { "content-type": "application/json", accept: "text/event-stream" },
        body: JSON.stringify({
          chat_id: state.chatId,
          text: clean,
          model: state.model,
          mode: state.mode,
          reply_to: replyQuote,
        }),
        signal: controller.signal,
      });

      if (!res.ok || !res.body) {
        const data = await res.json().catch(() => ({}));
        if (res.status === 403 && data.error === "shield_blocked") {
          optimistic.remove();
          const mins = data.retry_after ? Math.ceil(data.retry_after / 60) : 0;
          toast(
            mins
              ? "Режим защиты заблокировал отправку: похоже на попытку взлома. Повтори через " + mins + " мин."
              : "Режим защиты заблокировал это сообщение: похоже на вредоносный код или взлом."
          );
        } else if (res.status === 429) {
          optimistic.remove();
          toast(
            data.error === "shield_rate_limited"
              ? "Режим защиты: слишком много сообщений подряд. Подожди пару минут."
              : "Слишком много сообщений за час. Немного отдохни и продолжи."
          );
        } else {
          appendMessage({ role: "ai", who: "siri", text: "Облако сейчас не отвечает. Попробуй ещё раз через минуту." });
        }
        return;
      }

      setCloud(true);
      const reader = res.body.getReader();
      const decoder = new TextDecoder();
      let buffer = "";

      for (;;) {
        const { done, value } = await reader.read();
        if (done) break;
        buffer += decoder.decode(value, { stream: true });
        const blocks = buffer.split(/\r?\n\r?\n/);
        buffer = blocks.pop();
        for (const block of blocks) {
          const line = block.split("\n").find((l) => l.startsWith("data:"));
          if (!line) continue;
          let event;
          try { event = JSON.parse(line.slice(5).trim()); } catch { continue; }

          if (event.type === "start") {
            if (!state.chatId) state.chatId = event.chat.id;
            if (!state.chats.some((ch) => ch.id === event.chat.id)) {
              state.chats.unshift({
                id: event.chat.id,
                title: event.chat.title,
                preview: "",
                mode: event.chat.mode,
                updated_at: Date.now(),
              });
              renderChats();
            }
          } else if (event.type === "status") {
            if (!searchBanner) searchBanner = appendSearchBanner(event.text);
          } else if (event.type === "sources") {
            if (searchBanner) {
              searchBanner.textContent = "Нашёл в интернете: " + (event.query || "") + " — " + event.sources.length + " источн.";
            }
          } else if (event.type === "who") {
            sawFirst = true;
            const live = liveMessage(event.who);
            lives.set(event.who, live);
            paintLive(live, event.who, () => {
              const final = finals.get(event.who);
              if (final) {
                finishLive(live, final);
                speakIfLive(final);
              }
            });
          } else if (event.type === "delta") {
            const painter = [...painters].find((p) => p.live === lives.get(event.who));
            if (painter) painter.full += event.text;
          } else if (event.type === "saved") {
            const painter = [...painters].find((p) => p.live === lives.get(event.who));
            if (painter) painter.finished = true;
            finals.set(event.who, event.reply);
          } else if (event.type === "error") {
            const live = lives.get(event.who);
            const painter = [...painters].find((p) => p.live === live);
            if (painter) painter.finished = true;
            finals.set(event.who, {
              text: event.error === "save_failed"
                ? "Ответ пришёл, но сохранить его не удалось. Скопируй текст, если он нужен."
                : "Облако сейчас не отвечает. Попробуй ещё раз через минуту.",
              sources: [],
            });
            if (!live && !sawFirst) {
              appendMessage({ role: "ai", who: event.who || "siri", text: "Облако сейчас не отвечает. Попробуй ещё раз через минуту." });
            }
          } else if (event.type === "done") {
            for (const painter of painters) painter.finished = true;
          }
        }
      }

      // Всё, что не докрасилось, финализируем принудительно.
      await new Promise((resolve) => {
        const check = () => {
          if (painters.size === 0) return resolve();
          for (const painter of painters) painter.finished = true;
          requestAnimationFrame(check);
        };
        check();
      });
      for (const [who, live] of lives) {
        const final = finals.get(who);
        if (final) finishLive(live, final);
      }
      if (!lives.size && !sawFirst) {
        appendMessage({ role: "ai", who: "siri", text: "Облако сейчас не отвечает. Попробуй ещё раз через минуту." });
      }
      await loadChats();
    } catch {
      if (controller.signal.aborted) return;
      setCloud(false);
      appendMessage({ role: "ai", who: "siri", text: "Пропала связь с облаком. Проверь интернет и попробуй снова." });
    } finally {
      const current = streamAbort === controller;
      if (current) streamAbort = null;
      if (current || !streamAbort) {
        state.sending = false;
        sendBtn.disabled = false;
        scrollDown(false);
        updateJumpDown();
        inputEl.focus();
      }
    }
  }

  function setCloud(ok) {
    const el = $("cloudStatus");
    if (!el) return;
    const dot = document.createElement("span");
    dot.className = "dot";
    if (!ok) dot.style.background = "var(--error)";
    el.replaceChildren(dot, ok ? " в облаке" : " нет связи");
  }

  $("composer").addEventListener("submit", (e) => {
    e.preventDefault();
    send(inputEl.value);
  });

  inputEl.addEventListener("keydown", (e) => {
    if (e.key === "Enter" && !e.shiftKey) {
      e.preventDefault();
      send(inputEl.value);
    }
  });

  inputEl.addEventListener("input", () => {
    inputEl.style.height = "auto";
    inputEl.style.height = Math.min(inputEl.scrollHeight, 160) + "px";
  });

  $("newChat").addEventListener("click", resetView);
  $("topNewChat").addEventListener("click", resetView);
  $("chatSearch").addEventListener("input", (e) => {
    state.filter = e.target.value.trim().toLowerCase();
    renderChats();
  });

  document.querySelectorAll("[data-prompt]").forEach((btn) => {
    btn.addEventListener("click", () => send(btn.dataset.prompt));
  });

  /* ---------- плюс: быстрый поиск в интернете ---------- */
  $("plusBtn").addEventListener("click", () => {
    inputEl.value = "Поищи в интернете: " + inputEl.value;
    inputEl.focus();
  });

  /* ---------- запуск ---------- */
  setMode(store.get("siri-mode") || "siri", false);
  applyModel(store.get("siri-model") || "cyclone");
  state.speed = store.get("siri-speed") || "slow";
  if (speedSelect) speedSelect.value = state.speed;

  const whenIdle = window.requestIdleCallback || ((fn) => setTimeout(fn, 700));
  whenIdle(async () => {
    await loadSettingsIntoModal();
    await applyLightning($("lightningSwitch") ? $("lightningSwitch").checked : true);
  });

  (async () => {
    await loadChats();
    if (state.chats.length) selectChat(state.chats[0].id);
    else if (window.matchMedia("(min-width: 900px)").matches) inputEl.focus();
  })();
})();

(() => {
  const holder = document.querySelector("[data-full]");
  const showBtn = document.getElementById("sourceShow");
  const copyBtn = document.getElementById("sourceCopy");
  const status = document.getElementById("sourceStatus");
  const pre = document.getElementById("sourceFull");
  const codeEl = document.getElementById("sourceFullCode");
  const url = holder ? holder.dataset.full : "";
  if (!url || !showBtn || !copyBtn || !status || !pre || !codeEl) return;

  let fullCode = null;
  let loading = null;

  function setStatus(text) {
    status.textContent = text;
    status.hidden = !text;
  }

  async function load() {
    if (fullCode !== null) return fullCode;
    if (loading) return loading;
    setStatus("Загружаю весь код…");
    loading = fetch(url, { headers: { accept: "text/plain" } })
      .then((res) => {
        if (!res.ok) throw new Error("failed");
        return res.text();
      })
      .then((text) => {
        fullCode = text;
        setStatus("");
        return text;
      })
      .catch(() => {
        setStatus("Не удалось загрузить код. Откройте его в новой вкладке или скачайте файлом.");
        throw new Error("failed");
      })
      .finally(() => {
        loading = null;
      });
    return loading;
  }

  showBtn.addEventListener("click", async () => {
    if (!pre.hidden) {
      pre.hidden = true;
      showBtn.textContent = "Показать весь код";
      return;
    }
    showBtn.disabled = true;
    try {
      codeEl.textContent = await load();
      pre.hidden = false;
      showBtn.textContent = "Скрыть код";
      const reduce = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
      pre.scrollIntoView({ behavior: reduce ? "auto" : "smooth", block: "start" });
    } catch {
      /* the status line explains what to do next */
    } finally {
      showBtn.disabled = false;
    }
  });

  function legacyCopy(text) {
    const area = document.createElement("textarea");
    area.value = text;
    area.setAttribute("readonly", "");
    area.style.position = "fixed";
    area.style.opacity = "0";
    document.body.append(area);
    area.select();
    const ok = document.execCommand("copy");
    area.remove();
    return ok;
  }

  copyBtn.addEventListener("click", async () => {
    const label = copyBtn.textContent;
    copyBtn.disabled = true;
    try {
      if (fullCode === null) await load();
      try {
        await navigator.clipboard.writeText(fullCode);
      } catch {
        if (!legacyCopy(fullCode)) throw new Error("failed");
      }
      copyBtn.textContent = "Скопировано ✓";
    } catch {
      copyBtn.textContent = "Не удалось — скачайте файлом";
    }
    setTimeout(() => {
      copyBtn.textContent = label;
      copyBtn.disabled = false;
    }, 2400);
  });

  // Warm the code in the background so a tap copies instantly (iOS keeps the tap gesture only that way).
  const idle = window.requestIdleCallback || ((fn) => setTimeout(fn, 1200));
  idle(() => { load().catch(() => {}); });
})();
